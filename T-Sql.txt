CREATE DATABASE gwapi    CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
CREATE TABLE data_measure (
    id              INT             UNSIGNED            NOT NULL    AUTO_INCREMENT  COMMENT '流水號',
    sid             VARCHAR(50)                         NOT NULL                    COMMENT '資料唯一識別碼SID',
    facility_id     INT             UNSIGNED            NOT NULL                    COMMENT '所屬機構流水號，對應data_facility.id',
    measure_no      VARCHAR(100)                        NOT NULL                    COMMENT '健檢流水號',
    device_no       VARCHAR(100)                        NOT NULL                    COMMENT '設備編號',
    machine_model   VARCHAR(100)                        NOT NULL                    COMMENT '機器型號',
    online_type     VARCHAR(10)     DEFAULT 'ON-LINE'   NOT NULL                    COMMENT '版本類型：ON-LINE(線上版)、OFF-LINE(離線版)',
    is_uploaded     TINYINT(1)      DEFAULT 0           NOT NULL                    COMMENT '離線版專用：是否已同步/上傳至雲端主機(0: 未上傳, 1: 已上傳)',
    json_data       LONGTEXT        DEFAULT                 NULL                    COMMENT '健檢結果JSON資料',
    raw_data        LONGTEXT        DEFAULT                 NULL                    COMMENT '設備原始RAW DATA資料，內容格式為ASCII文字',
    file_name       VARCHAR(255)    DEFAULT                 NULL                    COMMENT '上傳檔案名稱，格式為日期_健檢流水號',
    mime_type       VARCHAR(100)    DEFAULT                 NULL                    COMMENT '上傳檔案MIME類型',
    file_size       BIGINT UNSIGNED DEFAULT 0           NOT NULL                    COMMENT '上傳檔案大小，單位Bytes',
    file_data       LONGBLOB                                NULL                    COMMENT '上傳檔案Binary二進位內容',
    measure_date    DATETIME                            NOT NULL                    COMMENT '健檢檢測日期時間',
    up_json_data    LONGTEXT        DEFAULT                 NULL                    COMMENT '共通JSON資料',
    remark          TEXT            DEFAULT                 NULL                    COMMENT '備註',
    created_at      DATETIME                            NOT NULL                    COMMENT '資料建立日期時間',
    updated_at      DATETIME        DEFAULT                 NULL                    COMMENT '資料最後更新日期時間',
    PRIMARY KEY (id),
    UNIQUE KEY uk_sid (sid),
    KEY idx_measure_no (measure_no),
    KEY idx_device_no (device_no),
    KEY idx_machine_model (machine_model),
    KEY idx_measure_date (measure_date),
    KEY idx_online_type (online_type),
    KEY idx_is_uploaded (is_uploaded)
) COMMENT='設備健檢量測資料表';

CREATE TABLE log_measure (
    id              BIGINT          UNSIGNED            NOT NULL    AUTO_INCREMENT  COMMENT 'LOG流水號',
    facility_id     INT             UNSIGNED            NOT NULL                    COMMENT '所屬機構流水號，對應data_facility.id',
    measure_id      INT             UNSIGNED DEFAULT        NULL                    COMMENT '對應data_measure流水號',
    sid             VARCHAR(50)                         NOT NULL                    COMMENT 'data_measure資料唯一識別碼SID',
    measure_no      VARCHAR(100)    DEFAULT                 NULL                    COMMENT '健檢流水號',
    device_no       VARCHAR(100)    DEFAULT                 NULL                    COMMENT '設備編號',
    machine_model   VARCHAR(100)    DEFAULT                 NULL                    COMMENT '機器型號',
    action_type     VARCHAR(20)                         NOT NULL                    COMMENT '異動類型：UPDATE、DELETE、REUPLOAD、SYNC、OTHER',
    change_data     LONGTEXT        DEFAULT                 NULL                    COMMENT '異動欄位內容JSON，記錄異動前後資料',
    action_user     VARCHAR(100)    DEFAULT                 NULL                    COMMENT '異動操作者帳號或識別碼',
    action_ip       VARCHAR(50)     DEFAULT                 NULL                    COMMENT '異動來源IP位址',
    action_note     VARCHAR(255)    DEFAULT                 NULL                    COMMENT '異動說明',
    created_at      DATETIME                            NOT NULL                    COMMENT 'LOG建立日期時間',
    PRIMARY KEY (id),
    KEY idx_measure_id (measure_id),
    KEY idx_sid (sid),
    KEY idx_measure_no (measure_no),
    KEY idx_device_no (device_no),
    KEY idx_action_type (action_type),
    KEY idx_created_at (created_at)
) COMMENT='設備健檢量測資料異動LOG表';

CREATE TABLE data_facility (
    id              INT             UNSIGNED    NOT NULL    AUTO_INCREMENT  COMMENT '機構資料流水號',

    facility_no     VARCHAR(100)                 NOT NULL                   COMMENT '外部系統機構編號',

    facility_name   VARCHAR(200)                NOT NULL                    COMMENT '機構名稱',

    created_at      DATETIME                    NOT NULL                    COMMENT '資料建立日期時間',

    updated_at      DATETIME        DEFAULT         NULL                    COMMENT '資料最後更新日期時間',

    PRIMARY KEY (id),

    UNIQUE KEY uk_facility_no (facility_no),

    KEY idx_facility_name (facility_name),

    KEY idx_status (status)

) COMMENT='機構資料表';

CREATE TABLE data_member (
    id              INT             UNSIGNED    NOT NULL    AUTO_INCREMENT  COMMENT '會員資料流水號',

    sid             VARCHAR(50)                 NOT NULL                    COMMENT '會員資料唯一識別碼SID',

    account         VARCHAR(100)                NOT NULL                    COMMENT '會員登入帳號',

    password        VARCHAR(255)                NOT NULL                    COMMENT '會員登入密碼雜湊值',

    member_name     VARCHAR(100)                NOT NULL                    COMMENT '會員姓名',

    facility_id     INT             UNSIGNED    NOT NULL                    COMMENT '所屬機構流水號，對應data_facility.id',

    role            VARCHAR(20)                 NOT NULL                    COMMENT '會員權限：superuser、admin',

    email           VARCHAR(255)    DEFAULT         NULL                    COMMENT '會員電子郵件',

    phone           VARCHAR(50)     DEFAULT         NULL                    COMMENT '會員聯絡電話',

    status          TINYINT         UNSIGNED    NOT NULL    DEFAULT 1       COMMENT '會員狀態：1啟用、0停用',

    last_login_at   DATETIME        DEFAULT         NULL                    COMMENT '最後登入日期時間',

    remark          VARCHAR(255)    DEFAULT         NULL                    COMMENT '備註',

    created_at      DATETIME                    NOT NULL                    COMMENT '資料建立日期時間',

    updated_at      DATETIME        DEFAULT         NULL                    COMMENT '資料最後更新日期時間',

    PRIMARY KEY (id),

    UNIQUE KEY uk_sid (sid),

    UNIQUE KEY uk_account (account),

    KEY idx_facility_id (facility_id),

    KEY idx_role (role),

    KEY idx_status (status),

    CONSTRAINT fk_member_facility
        FOREIGN KEY (facility_id)
        REFERENCES data_facility (id)

) COMMENT='會員資料表';

CREATE TABLE data_pc (
    id              INT             UNSIGNED    NOT NULL    AUTO_INCREMENT  COMMENT 'PC資料流水號',

    sid             VARCHAR(50)                 NOT NULL                    COMMENT 'PC資料唯一識別碼SID',

    mac_address     VARCHAR(17)                 NOT NULL                    COMMENT 'PC網路卡MAC位址，格式例如AA:BB:CC:DD:EE:FF',

    pc_name         VARCHAR(100)    DEFAULT         NULL                    COMMENT 'PC電腦名稱',

    description     VARCHAR(255)    DEFAULT         NULL                    COMMENT 'PC用途或說明',

    status          TINYINT         UNSIGNED    NOT NULL    DEFAULT 1       COMMENT '啟用狀態：1啟用、0停用',

    remark          VARCHAR(255)    DEFAULT         NULL                    COMMENT '備註',

    created_at      DATETIME                    NOT NULL                    COMMENT '資料建立日期時間',

    updated_at      DATETIME        DEFAULT         NULL                    COMMENT '資料最後更新日期時間',

    PRIMARY KEY (id),

    UNIQUE KEY uk_sid (sid),

    UNIQUE KEY uk_mac_address (mac_address),

    KEY idx_pc_name (pc_name),

    KEY idx_status (status)

) COMMENT='PC設備資料表';
CREATE TABLE log_pc (
    id              BIGINT          UNSIGNED    NOT NULL    AUTO_INCREMENT  COMMENT 'PC資料LOG流水號',

    pc_id           INT             UNSIGNED    DEFAULT         NULL                    COMMENT '對應data_pc流水號',

    sid             VARCHAR(50)     DEFAULT         NULL                    COMMENT 'PC資料唯一識別碼SID',

    mac_address     VARCHAR(17)     DEFAULT         NULL                    COMMENT 'PC網路卡MAC位址',

    pc_name         VARCHAR(100)    DEFAULT         NULL                    COMMENT 'PC電腦名稱',

    action_type     VARCHAR(20)                 NOT NULL                    COMMENT '異動類型：INSERT、UPDATE、DELETE',

    change_data     LONGTEXT        DEFAULT         NULL                    COMMENT '異動欄位JSON資料',

    action_user     VARCHAR(100)    DEFAULT         NULL                    COMMENT '異動操作者帳號或識別碼',

    action_ip       VARCHAR(50)     DEFAULT         NULL                    COMMENT '異動來源IP位址',

    action_note     VARCHAR(255)    DEFAULT         NULL                    COMMENT '異動說明',

    created_at      DATETIME                    NOT NULL                    COMMENT 'LOG建立日期時間',

    PRIMARY KEY (id),

    KEY idx_pc_id (pc_id),

    KEY idx_sid (sid),

    KEY idx_mac_address (mac_address),

    KEY idx_action_type (action_type),

    KEY idx_created_at (created_at)

) COMMENT='PC設備資料異動LOG表';
